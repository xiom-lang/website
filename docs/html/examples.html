  "kw">return x * x + y * y;

7. Error Handling — Result & ?


"kw">fn divide_safe(a: "ty">Float64, b: "ty">Float64) -> "ty">Result["ty">Float64, "ty">Str] {
  "kw">if b == 0.0 {
    "kw">return "kw">Err("division by zero");
  }
  "kw">return "kw">Ok(a / b);
}

"kw">fn compute(x: "ty">Float64, y: "ty">Float64) -> "ty">Result["ty">Float64, "ty">Str] {
  "kw">let step1 = divide_safe(x, y)?;    "cm">// ? propagates error
  "kw">let step2 = divide_safe(step1, 2.0)?;
  "kw">return "kw">Ok(step2);
}

"kw">fn main() -> "ty">Int {
  "kw">match compute(10.0, 2.0) {
    "kw">Ok("kw">result) => "kw">result.to_str(),
    "kw">Err(msg)   => msg,
  };
  "kw">return 0;
}

? returns the error immediately. Only usable inside functions returning Result or Option.

8. Contracts — Divide by Zero Prevention


"kw">fn divide(a: "ty">Float64, b: "ty">Float64) -> "ty">Float64
  "kw">requires: b != 0.0
  "kw">ensures:  "kw">result * b == a
{
  "kw">return a / b;
}

"kw">type PositiveInt = {
  value: "ty">Int;
  "kw">invariant: value > 0;
}

"kw">fn main() -> "ty">Int {
  "kw">let x = divide(10.0, 2.0);   "cm">// ok
  "cm">// "kw">let y = divide(10.0, 0.0); // COMPILE ERROR: contract violated
  "kw">return 0;
}

Contracts are compiler-enforced. requires = caller's responsibility. ensures = implementation's responsibility. invariant = always true after any mutation.

9. Ownership & Borrowing


"kw">fn take_ownership(x: "ty">Int) -> "ty">Int {
  "kw">return x + 1;                    "cm">// x freed at end of scope
}

"kw">fn read_borrow(x: &"ty">Int) -> "ty">Int {
  "kw">return x + 0;                    "cm">// caller retains ownership
}

"kw">fn main() -> "ty">Int {
  "kw">let a = take_ownership(41);      "cm">// a = 42
  "kw">let b = read_borrow(&a);         "cm">// borrow — a still valid
  "kw">return a + b;                    "cm">// 84
}
  • &T = read borrow (multiple allowed)
  • &mut T = write borrow (exclusive)
  • Passing without & = move ownership (old binding invalid)
  • .clone() = explicit deep copy

11. Interfaces — Comparable


"kw">interface "ty">Comparable {
  "kw">fn compare(other: &"ty">Self) -> "ty">Int;
}

"kw">type "ty">Score = { value: "ty">Int; }

"kw">fn "ty">Score.compare(other: &"ty">Score) -> "ty">Int {
  "kw">if value < other.value { "kw">return -1; }
  "kw">if value > other.value { "kw">return  1; }
  "kw">return 0;
}

"kw">fn max[T: "ty">Comparable](a: T, b: T) -> T {
  "kw">if a.compare(&b) > 0 { "kw">return a; }
  "kw">return b;
}

"kw">fn main() -> "ty">Int {
  "kw">let s1 = "ty">Score{ value: 10 };
  "kw">let s2 = "ty">Score{ value: 20 };
  "kw">let winner = max(s1, s2);
  "kw">return winner.value;   "cm">// 20
}

No implements keyword. A type satisfies an interface by having the right methods.

12. Sort with Contracts


"kw">fn sort(items: &mut "ty">Vec["ty">Int])
  "kw">ensures: items.is_sorted()
{
  "cm">// bubble sort "kw">for clarity
  "kw">let n = items.len();
  "kw">var i = 0;
  "kw">while i < n {
    "kw">var j = 0;
    "kw">while j < n - i - 1 {
      "kw">if items.get(j).unwrap() > items.get(j + 1).unwrap() {
        "kw">let tmp = items.get(j).unwrap();
        items.set(j, items.get(j + 1).unwrap());
        items.set(j + 1, tmp);
      }
      j = j + 1;
    }
    i = i + 1;
  }
}

"kw">fn main() -> "ty">Int {
  "kw">var nums = [4, 1, 3, 2];
  sort(&mut nums);
  "cm">// nums "kw">is now [1, 2, 3, 4] — verified by contract
  "kw">return 0;
}

The ensures: items.is_sorted() contract is checked after the function returns. If the sort implementation is wrong, the contract violation surfaces immediately.
